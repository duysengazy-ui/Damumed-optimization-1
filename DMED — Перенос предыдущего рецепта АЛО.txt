// ==UserScript==
// @name         DMED — Перенос предыдущего рецепта АЛО
// @namespace    dmed-prescription-helper
// @version      1.6
// @description  Перенос параметров предыдущего рецепта в новый рецепт АЛО

(function () {
    'use strict';

    const STORAGE_KEY = 'dmed_previous_alo_recipe';

    // ------------------------------------------------------------
    // Утилиты
    // ------------------------------------------------------------

    function log(...args) {
        console.log('[DMED RX]', ...args);
    }

    function getGridFromRow(row) {
        return $(row).closest('.k-grid').data('kendoGrid');
    }

    function getRecipesFromCurrentPage() {
        const rows = [...document.querySelectorAll('tr')];

        const recipeRows = rows.filter(tr =>
            tr.innerText.includes('Дата выписки рецепта') &&
            tr.innerText.includes('Rp:')
        );

        const recipes = [];
        const seen = new Set();

        for (const recipeRow of recipeRows) {
            const grid = getGridFromRow(recipeRow);

            if (!grid) {
                continue;
            }

            const item = grid.dataItem(recipeRow);

            if (!item) {
                continue;
            }

            const recipe = item.toJSON
                ? item.toJSON()
                : { ...item };

            recipe.__rowText = String(recipeRow.innerText || '');

            const key = String(
                recipe.ID ||
                recipe.PrescriptionNumber ||
                `${recipe.RecipeDate || ''}|${recipe.Rp || ''}`
            );

            if (seen.has(key)) {
                continue;
            }

            seen.add(key);
            recipes.push(recipe);
        }

        return recipes;
    }

    function parseMnnFromRp(rp) {
        if (!rp) return '';

        // Например:
        // Оланзапин, 10 мг, Таблетки...
        return rp
            .split(',')[0]
            .trim();
    }

    function parseDose(recipe) {
        return String(
            recipe.Dosage ??
            ''
        ).trim();
    }

    function parseForm(recipe) {
        return String(
            recipe.DrugDosageForm ??
            ''
        ).trim()
            .toLowerCase();
    }

    // ------------------------------------------------------------
    // Работа с сохранёнными рецептами
    // ------------------------------------------------------------

    function getStoredRecipes() {
        try {
            const raw = localStorage.getItem(STORAGE_KEY);

            if (!raw) {
                return [];
            }

            const parsed = JSON.parse(raw);

            // Совместимость со старой версией,
            // где сохранялся только один объект
            if (Array.isArray(parsed)) {
                return parsed.map(recipe => {
                    const tradeName = String(
                        recipe.tradeName || ''
                    ).trim().toLowerCase();

                    if (tradeName === 'палиперидон') {
                        recipe.mnn = 'Палиперидон';
                    }

                    if (
                        tradeName === 'тизерцин' ||
                        tradeName === 'тизерцин 25 мг'
                    ) {
                        recipe.mnn = 'Левомепромазин';
                    }

                    return recipe;
                });
            }

            if (parsed && typeof parsed === 'object') {
                const recipe = parsed;

                const tradeName = String(
                    recipe.tradeName || ''
                ).trim().toLowerCase();

                if (tradeName === 'палиперидон') {
                    recipe.mnn = 'Палиперидон';
                }

                if (
                    tradeName === 'тизерцин' ||
                    tradeName === 'тизерцин 25 мг'
                ) {
                    recipe.mnn = 'Левомепромазин';
                }

                return [recipe];
            }

            return [];
        } catch (e) {
            console.error('[DMED RX] Ошибка чтения рецептов:', e);
            return [];
        }
    }

    function getStoredRecipe() {
        const recipes = getStoredRecipes();

        if (!recipes.length) {
            return null;
        }

        return recipes[recipes.length - 1];
    }

    function saveRecipe(recipe) {
        const recipes = getStoredRecipes();

        const key = [
            recipe.mnn || '',
            recipe.dosage || '',
            recipe.tradeName || ''
        ]
            .join('|')
            .toLowerCase();

        const existingIndex = recipes.findIndex(item => {
            const itemKey = [
                item.mnn || '',
                item.dosage || '',
                item.tradeName || ''
            ]
                .join('|')
                .toLowerCase();

            return itemKey === key;
        });

        if (existingIndex >= 0) {
            recipes[existingIndex] = recipe;
        } else {
            recipes.push(recipe);
        }

        localStorage.setItem(
            STORAGE_KEY,
            JSON.stringify(recipes)
        );
    }

    function parseRecipeDate(value) {
        if (value instanceof Date) return value.getTime();
        if (typeof value === 'number') return value;

        const text = String(value || '').trim();
        const microsoft = text.match(/\/Date\((\d+)/i);
        if (microsoft) return Number(microsoft[1]);

        const ru = text.match(
            /(\d{1,2})\.(\d{1,2})\.(\d{4})(?:\s+(\d{1,2}):(\d{2}))?/
        );

        if (ru) {
            return new Date(
                Number(ru[3]), Number(ru[2]) - 1, Number(ru[1]),
                Number(ru[4] || 0), Number(ru[5] || 0)
            ).getTime();
        }

        const parsed = Date.parse(text);
        return Number.isFinite(parsed) ? parsed : 0;
    }

    function firstPresent(...values) {
        return values.find(value =>
            value != null && String(value).trim() !== ''
        );
    }

    function extractUseMethod(recipe) {
        const text = String(recipe.__rowText || '');
        const ds = String(recipe.Ds ?? recipe.ds ?? '');

        return String(firstPresent(
            recipe.UseMethodName,
            recipe.UseMethod?.Name,
            recipe.UseMethod?.NameRU,
            recipe.RouteName,
            text.match(/Способ\s+введения\s*:\s*([^\r\n]+)/i)?.[1],
            ds.match(/^\s*([^,;]+)/)?.[1]
        ) || '').trim();
    }

    function extractDtd(recipe) {
        const text = String(recipe.__rowText || '');

        return firstPresent(
            recipe.Dtd,
            recipe.DTD,
            recipe.dtd,
            recipe.NumberOfPackages,
            recipe.Quantity,
            text.match(/D\.?\s*t\.?\s*d\.?\s*:\s*([\d.,]+)/i)?.[1]
        );
    }

    function getLatestStoredRecipeGroup() {
        const recipes = getStoredRecipes();

        if (!recipes.length) {
            return [];
        }

        const dated = recipes
            .filter(recipe =>
                String(recipe.tradeName || recipe.mnn || recipe.rp || '').trim()
            )
            .map((recipe, index) => ({
                recipe,
                index,
                date: parseRecipeDate(recipe.recipeDate)
            }))
            .sort((a, b) => b.date - a.date || a.index - b.index);

        if (!dated.length) return [];

        const latestDate = dated[0].date;

        // Все препараты одного последнего заполненного визита.
        // Если последний месяц был пустым, его в хранилище нет,
        // поэтому автоматически берётся предыдущий непустой месяц.
        return dated
            .filter(item => item.date === latestDate)
            .map(item => item.recipe);
    }

    function chooseStoredRecipe() {
        const recipes = getStoredRecipes();
        if (!recipes.length) return null;

        const list = recipes.map((recipe, index) =>
            `${index + 1}. ${getRecipeSearchText(recipe) || 'Препарат'} ` +
            `— ${recipe.dosageConcentration ?? recipe.dosage ?? 'доза не указана'} ` +
            `— ${recipe.recipePackQty ?? '?'} уп.`
        ).join('\n');

        const answer = prompt(
            'Выбери предыдущий рецепт:\n\n' + list + '\n\nВведи номер:'
        );

        if (answer === null) return null;

        const index = Number(answer) - 1;
        if (!Number.isInteger(index) || index < 0 || index >= recipes.length) {
            alert('Неверный номер рецепта.');
            return null;
        }

        return {
            recipe: recipes[index],
            position: index + 1,
            total: recipes.length
        };
    }

    function getRecipeSearchText(recipe) {
        const source = [
            recipe.tradeName,
            recipe.mnn,
            recipe.rp,
            recipe.Rp,
            recipe.dosageForm
        ].filter(Boolean).join(' ').toLowerCase();

        if (
            source.includes('рисполепт конста') ||
            source.includes('rispolept consta') ||
            (source.includes('рисперидон') && source.includes('инъек'))
        ) {
            return 'Конста';
        }

        if (source.includes('галоперидол') || source.includes('haloperidol')) {
            if (
                source.includes('деканоат') ||
                source.includes('decanoate') ||
                source.includes('масля') ||
                source.includes('инъек')
            ) {
                return 'Деканоат';
            }

            return 'Haloperidol';
        }

        return String(recipe.tradeName || '').trim();
    }

    function visible(element) {
        return !!(
            element &&
            element.getClientRects().length &&
            getComputedStyle(element).visibility !== 'hidden'
        );
    }

    async function waitFor(test, timeout = 6000) {
        const end = Date.now() + timeout;

        while (Date.now() < end) {
            const result = test();
            if (result) return result;
            await new Promise(resolve => setTimeout(resolve, 100));
        }

        return null;
    }

    async function ensureAloRecipeForm() {
        let search = document.querySelector('#tbSearchFreeMedication');
        if (search) return search;

        const aloTab = [...document.querySelectorAll('a, button, span, div')]
            .find(element =>
                visible(element) &&
                String(element.textContent || '')
                    .replace(/\s+/g, ' ')
                    .trim()
                    .toLowerCase() === 'рецепты ало'
            );

        if (aloTab) {
            aloTab.click();
            await new Promise(resolve => setTimeout(resolve, 300));
        }

        search = await waitFor(() =>
            document.querySelector('#tbSearchFreeMedication'), 1200
        );
        if (search) return search;

        const addButton = [...document.querySelectorAll(
            '.k-grid-add, [title*="Добав" i], [aria-label*="Добав" i], a, button'
        )].find(element => {
            if (!visible(element)) return false;
            const text = String(element.textContent || '').trim();
            return element.matches('.k-grid-add, [title*="Добав" i], [aria-label*="Добав" i]') ||
                text === '+';
        });

        if (addButton) addButton.click();

        return await waitFor(() =>
            document.querySelector('#tbSearchFreeMedication'), 6000
        );
    }

    function getTradeNameFallback(recipe) {
        const mnn = String(
            recipe.mnn ||
            parseMnnFromRp(recipe.Rp) ||
            ''
        )
            .trim()
            .toLowerCase();

        const dosage = String(
            recipe.Dosage ??
            ''
        ).trim();

        const form = parseForm(recipe);

        const rp = String(
            recipe.Rp ||
            ''
        ).trim().toLowerCase();

        // Клозапин
        if (mnn.includes('клозапин')) {
            return 'Азалептол';
        }

        // Галоперидол деканоат
        if (
            mnn.includes('галоперидол') &&
            (
                rp.includes('деканоат') ||
                form.includes('масля') ||
                form.includes('инъек')
            )
        ) {
            return 'Деканоат';
        }

        // Венлафаксин
        if (mnn.includes('венлафаксин')) {
            return 'Алвента';
        }

        // Амисульприд
        if (mnn.includes('амисульприд')) {
            return 'Амисульприд Санто';
        }

        // Хлорпромазин
        if (mnn.includes('хлорпромазин')) {
            return 'Аминазин';
        }

        // Амитриптилин
        if (mnn.includes('амитриптилин')) {
            return 'Амитриптилин-ТК';
        }

        // Обычный галоперидол
        if (mnn.includes('галоперидол')) {
            return 'Haloperidol';
        }

        // Ксеплион
        if (
            mnn.includes('палиперидон') &&
            (
                dosage === '100' ||
                dosage === '150' ||
                form.includes('инъек')
            )
        ) {
            return 'Ксеплион';
        }

        // Олфрекс
        if (mnn.includes('оланзапин')) {
            return 'Олфрекс';
        }

        // Палиперидон в таблетках
        if (mnn.includes('палиперидон')) {
            return 'Палиперидон';
        }

        // Рисполепт Конста
        if (
            mnn.includes('рисперидон') &&
            (
                dosage === '25' ||
                dosage === '37.5' ||
                dosage === '37,5' ||
                rp.includes('конста')
            )
        ) {
            if (
                dosage === '37.5' ||
                dosage === '37,5' ||
                rp.includes('37,5')
            ) {
                return 'Конста';
            }

            return 'Конста';
        }

        // Рисперидон
        if (mnn.includes('рисперидон')) {
            return 'Рисперидон';
        }

        // Тизерцин / левомепромазин
        if (mnn.includes('левомепромазин')) {
            return 'Тизерцин';
        }

        // Тревикта
        if (
            mnn.includes('палиперидон') &&
            (
                dosage === '350' ||
                dosage === '525'
            )
        ) {
            if (dosage === '525') {
                return 'Тревикта 525 мг';
            }

            return 'Тревикта 350 мг';
        }

        // Трифтазин-Здоровье / трифлуоперазин
        if (
            mnn.includes('трифлуоперазин')
        ) {
            return 'Трифтазин-Здоровье';
        }

        // Циклодол / тригексифенидил
        if (
            mnn.includes('тригексифенидил') ||
            mnn.includes('тригексифенидила')
        ) {
            return 'Циклодол';
        }

        return null;
    }

    function getTradeNameFromPage(recipe) {

        // --------------------------------------------------------
        // Сначала определяем торговое название по МНН.
        // Это не позволяет взять чужой препарат с текущей страницы.
        // --------------------------------------------------------

        const fallback = getTradeNameFallback(recipe);

        if (fallback) {
            log(
                'Торговое название определено по МНН:',
                fallback
            );

            return fallback;
        }

        const html = document.documentElement.innerHTML;

        // Ищем непосредственно в HTML страницы.
        // DMED может использовать:
        // обычный пробел
        // &nbsp;
        // &amp;nbsp;

        const space = '(?:\\s|&nbsp;|&amp;nbsp;)*';

        const match = html.match(
            new RegExp(
                'Таб\\.' +
                space +
                '([^<]+?)' +
                space +
                '\\d+(?:[.,]\\d+)?' +
                space +
                'мг' +
                space +
                '№' +
                space +
                '\\d+',
                'i'
            )
        );

        if (match) {
            const tradeName = match[1]
                .replace(/&nbsp;/gi, ' ')
                .replace(/&amp;/gi, '&')
                .replace(/\s+/g, ' ')
                .trim();

            log('Торговое название найдено:', tradeName);

            return tradeName;
        }

        log('Торговое название по старому шаблону не найдено');

        // --------------------------------------------------------
        // ДОБАВЛЕНО:
        // Для растворов, инъекций и других форм используем
        // соответствие МНН → торговое название.
        // Старую рабочую логику выше не трогаем.
        // --------------------------------------------------------

        const fallback2 = getTradeNameFallback(recipe);

        if (fallback2) {
            log(
                'Торговое название определено по МНН:',
                fallback2
            );

            return fallback2;
        }

        // Диагностика
        const pos = html.toLowerCase().indexOf('таб.');

        if (pos !== -1) {
            log(
                'Фрагмент HTML около Таб.:',
                html.substring(pos, pos + 500)
            );
        }

        return null;
    }

    function rememberPreviousRecipe() {
        const recipes = getRecipesFromCurrentPage();

        console.log('[DMED RX] RAW RECIPES:', recipes);

        if (!recipes.length) {
            alert(
                'Не найден рецепт на текущей странице.\n\n' +
                'Открой предыдущую запись → Рецепты АЛО → убедись, ' +
                'что рецепты отображаются.'
            );

            return;
        }

        const saved = [];

        for (const recipe of recipes) {
            const mnn = parseMnnFromRp(recipe.Rp);

            const simplified = {
                ID: recipe.ID,

                // МНН — для информации
                mnn: mnn,

                // Торговое название — именно его будем вводить в поиск DMED
                tradeName: getTradeNameFromPage({
                    ...recipe,
                    mnn: mnn
                }),

                dosage: recipe.Dosage,
                dosageConcentration: recipe.DosageConcentration,

                dosageForm: recipe.DrugDosageForm,

                recipePackQty: firstPresent(
                    recipe.RecipePackQty,
                    recipe.PackageQuantity,
                    recipe.PackQty
                ),

                dtd: extractDtd(recipe),
                ds: recipe.Ds,

                useMethodName: extractUseMethod(recipe),

                rp: recipe.Rp,

                prescriptionNumber: recipe.PrescriptionNumber,

                recipeDate: recipe.RecipeDate
            };

            saveRecipe(simplified);
            saved.push(simplified);

            log('Сохранён предыдущий рецепт:', simplified);
        }

        const list = saved.map(recipe =>
            `${recipe.tradeName || recipe.rp || recipe.mnn} — ` +
            `${recipe.dosageConcentration ?? recipe.dosage ?? 'доза не указана'} — ` +
            `${recipe.recipePackQty ?? '?'} уп.`
        ).join('\n');

        alert(
            'Предыдущие рецепты сохранены.\n\n' +
            list
        );
    }

    function findAvailableDrugRow(recipe) {
        const mnn = String(recipe.mnn || '').trim().toLowerCase();

        // Для поиска используем реальную дозировку препарата.
        // В некоторых рецептах recipe.Dosage содержит количество
        // таблеток в упаковке (например, 25), а не дозировку препарата.
        const dosage = String(
    (
        String(recipe.rp || recipe.Rp || '').match(
            /,\s*([\d.,]+)\s*мг/i
        )?.[1] ||
        recipe.dosageConcentration ||
        recipe.dosage ||
        ''
    )
).trim();

        const rows = [...document.querySelectorAll('tr')];

        const candidates = rows.filter(row => {
            const grid = getGridFromRow(row);

            if (!grid) return false;

            const item = grid.dataItem(row);

            if (!item) return false;

            const data = item.toJSON
                ? item.toJSON()
                : item;

            const itemMnn = String(
                data.DrugMNNName || ''
            ).trim().toLowerCase();

            const itemDosage = String(
                data.Dosage ?? ''
            ).trim();

            const packages = Number(
                data.StockInPackages || 0
            );

            const units = Number(
                data.StockInUnits || 0
            );

            return (
                itemMnn === mnn &&
                itemDosage === dosage &&
                (packages > 0 || units > 0)
            );
        });

        if (!candidates.length) {
            return null;
        }

        return candidates[0];
    }

    function normalizeText(value) {
        return String(value || '')
            .toLowerCase()
            .replace(/ё/g, 'е')
            .replace(/[^a-zа-я0-9]+/gi, ' ')
            .replace(/\s+/g, ' ')
            .trim();
    }

    async function setUseMethod(name) {
        if (!name) {
            log('Способ введения не указан');
            return false;
        }

        const combo = $('#cmbFreeUseMethods')
            .data('kendoComboBox');

        if (!combo) {
            log('ComboBox способа введения не найден');
            return false;
        }

        let methods = [];

        for (let attempt = 0; attempt < 30; attempt++) {
            methods = combo.dataSource
                .data()
                .map(x => x.toJSON ? x.toJSON() : x);

            if (methods.length) break;

            if (attempt === 0) {
                try { combo.dataSource.read(); } catch (e) {}
            }

            await new Promise(resolve => setTimeout(resolve, 100));
        }

        const wanted = normalizeText(name);
        const methodName = x => normalizeText(
            x.Name || x.NameRu || x.DisplayName || x.Text
        );

        // Сначала ищем актуальный вариант:
        // EndDate == null
        let matchIndex = methods.findIndex(x =>
            methodName(x) === wanted &&
            !x.EndDate
        );

        // Если не нашли — любой совпадающий
        if (matchIndex < 0) {
            matchIndex = methods.findIndex(x =>
                methodName(x) === wanted
            );
        }

        // В разных версиях справочника встречаются уточнения в скобках.
        if (matchIndex < 0) {
            matchIndex = methods.findIndex(x =>
                methodName(x).includes(wanted) || wanted.includes(methodName(x))
            );
        }

        if (matchIndex < 0) {
            log('Не найден способ введения:', name);
            return false;
        }

        const match = methods[matchIndex];

        // Сначала выбираем именно строку ComboBox,
        // затем синхронизируем его значение и событие change.
        combo.select(matchIndex);
        combo.value(String(match.ID));
        combo.trigger('change');

        log(
            'Способ введения установлен:',
            match.ID,
            match.Name
        );

        return String(combo.value() || '') === String(match.ID) &&
            !!String(combo.text?.() || combo.input?.val?.() || '').trim();
    }

    function setPackages(value) {
        const input = document.querySelector(
            '#ntbNumberOfPackages'
        );

        if (!input) {
            log('Поле количества упаковок не найдено');
            return false;
        }

        const match = String(value ?? '')
            .replace(',', '.')
            .match(/\d+(?:\.\d+)?/);

        const quantity = match ? Number(match[0]) : NaN;

        if (!Number.isFinite(quantity) || quantity <= 0) {
            log('Некорректное количество:', value);
            return false;
        }

        const numeric = $(input).data('kendoNumericTextBox');

        if (numeric) {
            numeric.value(quantity);
            numeric.trigger('change');

            // Дополнительно синхронизируем обычное input-поле,
            // чтобы DMED не потерял значение при перерисовке формы.
            input.value = String(quantity);
            $(input).trigger('input');
            $(input).trigger('change');
        } else {
            input.value = String(quantity);
            $(input).trigger('change');
            $(input).trigger('input');
        }

        log(
            'Количество упаковок установлено:',
            quantity
        );

        const actual = numeric
            ? Number(numeric.value())
            : Number(String(input.value).replace(',', '.'));

        return Number.isFinite(actual) && actual === quantity;
    }

    // ------------------------------------------------------------
    // 5. Перенос рецепта
    // ------------------------------------------------------------

    async function applyPreviousRecipe() {
        const selection = chooseStoredRecipe();
        const recipe = selection?.recipe;

        if (!recipe) {
            if (!getStoredRecipes().length) {
                alert(
                    'Нет сохранённого предыдущего рецепта.\n\n' +
                    'Сначала открой предыдущую запись и нажми ' +
                    '«Запомнить предыдущий рецепт».'
                );
            }

            return;
        }

        log(
            `Применяем рецепт ${selection.position} из ${selection.total}:`,
            recipe
        );

        // Проверяем, что форма рецепта открыта
        const drugSearch = await ensureAloRecipeForm();

        if (!drugSearch) {
            alert(
                'Не удалось автоматически открыть форму «Рецепты АЛО».\n\n' +
                'Открой вкладку один раз и сообщи, если кнопка + выглядит иначе.'
            );

            return;
        }

        // --------------------------------------------------------
        // Выбираем категорию «Все категории, состоящие
        // на динамическом наблюдении»
        // --------------------------------------------------------

        const categoryCombo =
            $('#cmbPrescriptionCategory').data('kendoComboBox');

        if (categoryCombo) {
            categoryCombo.value('13');
            categoryCombo.trigger('change');

            log(
                'Категория установлена:',
                categoryCombo.value(),
                categoryCombo.text()
            );
        }

        // Ждём обновления таблицы остатков
        await new Promise(resolve =>
            setTimeout(resolve, 500)
        );

        // --------------------------------------------------------
        // Ищем препарат через штатный поиск DMED
        // --------------------------------------------------------

        const searchInput = document.querySelector(
            '#tbSearchFreeMedication'
        );

        if (!searchInput) {
            alert('Поле поиска препарата не найдено.');
            return;
        }

        const searchText = getRecipeSearchText(recipe);

        if (!searchText) {
            alert(
                'Не удалось определить торговое наименование препарата.\n\n' +
                'Открой предыдущую запись и убедись, что в тексте рецепта указано торговое название.'
            );

            return;
        }

        log('Ищем препарат через DMED:', searchText);

        // --------------------------------------------------------
        // Заполняем только поле поиска.
        // Кнопку поиска DMED НЕ нажимаем.
        // Ты нажимаешь её сам.
        // --------------------------------------------------------

        const searchWidget =
            $(searchInput).data('kendoAutoComplete') ||
            $(searchInput).data('kendoComboBox') ||
            $(searchInput).data('kendoDropDownList');

        if (searchWidget) {
            searchWidget.value(searchText);

            try {
                searchWidget.trigger('change');
                searchWidget.search?.(searchText);
            } catch (e) {}

            log(
                'Значение поиска установлено через Kendo:',
                searchText
            );
        } else {
            searchInput.value = searchText;
            searchInput.dispatchEvent(new Event('input', { bubbles: true }));
            searchInput.dispatchEvent(new Event('change', { bubbles: true }));

            log(
                'Kendo-контрол не найден, значение вставлено напрямую:',
                searchText
            );
        }

        // --------------------------------------------------------
        // Автоматически запускаем штатный поиск
        // --------------------------------------------------------

        const searchButton =
            document.querySelector('#btnSearchFreeMedication') ||
            searchInput.parentElement?.querySelector('button, .k-button, a');

        if (searchButton && visible(searchButton)) {
            searchButton.click();
        } else {
            searchInput.dispatchEvent(new KeyboardEvent('keydown', {
                key: 'Enter',
                code: 'Enter',
                keyCode: 13,
                which: 13,
                bubbles: true
            }));
        }

        log('Ждём результат автоматического поиска DMED...');

        let row = null;

        for (let i = 0; i < 10; i++) {
            row = findAvailableDrugRow(recipe);

            if (row) {
                log('Препарат найден автоматически');
                break;
            }

            await new Promise(resolve =>
                setTimeout(resolve, 500)
            );
        }

        if (!row) {
            alert(
                'Препарат не найден автоматическим поиском DMED.\n\n' +
                'Торговое наименование: ' +
                (recipe.tradeName || 'не определено') + '\n' +
                'МНН: ' +
                (recipe.mnn || 'не определено') + '\n' +
             'Дозировка: ' +
(
    recipe.dosageConcentration ||
    (
        String(recipe.rp || recipe.Rp || '').match(
            /,\s*([\d.,]+)\s*мг/i
        )?.[1] || ''
    ) ||
    recipe.dosage ||
    'не определена'
) + '\n\n' +
                'Проверь наличие препарата в остатках.'
            );

            return;
        }

        // --------------------------------------------------------
        // Обрабатываем найденную строку препарата
        // --------------------------------------------------------

        const grid = getGridFromRow(row);
        const item = grid.dataItem(row);

        const drug = item.toJSON
            ? item.toJSON()
            : item;

        log('Найден препарат:', drug);

        const stockPackages =
            Number(drug.StockInPackages || 0);

        const stockUnits =
            Number(drug.StockInUnits || 0);

        log(
            'Остаток:',
            stockPackages,
            'упаковок;',
            stockUnits,
            'единиц'
        );

        // --------------------------------------------------------
        // Кликаем по найденному препарату
        // --------------------------------------------------------

        row.click();

        $(row).trigger('click');

        // --------------------------------------------------------
        // ДОБАВЛЕНО:
        // DMED после клика может ещё заполнять форму.
        // Поэтому даём ему время и затем несколько раз
        // устанавливаем сохранённые параметры.
        // --------------------------------------------------------

        await new Promise(resolve =>
            setTimeout(resolve, 1000)
        );

        let methodSet = false;
        let packagesSet = false;

        for (let i = 0; i < 10; i++) {

            // ----------------------------------------------------
            // Способ введения
            // ----------------------------------------------------

            const savedMethod = firstPresent(
                recipe.useMethodName,
                String(recipe.ds || '').match(/^\s*([^,;]+)/)?.[1]
            );

            if (savedMethod) {
                methodSet = await setUseMethod(
                    savedMethod
                );
            } else {
                const methodCombo = $('#cmbFreeUseMethods').data('kendoComboBox');
                methodSet = !!String(
                    methodCombo?.value?.() || methodCombo?.text?.() || ''
                ).trim();
            }

            // Даём DMED время обработать change способа введения
            // до установки количества упаковок.
            await new Promise(resolve =>
                setTimeout(resolve, 150)
            );

            // ----------------------------------------------------
            // Количество упаковок
            // ----------------------------------------------------

            const quantity = firstPresent(
                recipe.dtd,
                recipe.Dtd,
                recipe.recipePackQty
            );

            if (quantity != null && String(quantity).trim() !== '') {
                packagesSet = setPackages(
                    quantity
                );
            } else {
                packagesSet = false;
            }

            if (methodSet && packagesSet) {
                log(
                    'Способ введения и количество успешно установлены'
                );

                break;
            }

            await new Promise(resolve =>
                setTimeout(resolve, 300)
            );
        }

        if (!methodSet || !packagesSet) {
            alert(
                'Препарат выбран, но не удалось перенести: ' +
                [
                    !methodSet ? 'способ введения' : '',
                    !packagesSet ? 'количество' : ''
                ].filter(Boolean).join(' и ') +
                '.\n\nПерезапомни этот рецепт из предыдущей записи: в старом сохранении этих данных может не быть.'
            );
            return;
        }

        log('Заполнение завершено и проверено');

        alert(
            'Параметры предыдущего рецепта перенесены.\n\n' +
            'Проверь форму перед сохранением.'
        );
    }

    // 6. UI
    // ------------------------------------------------------------

    function createButton(text, title, handler) {
        const button = document.createElement('button');

        button.type = 'button';
        button.textContent = text;

        button.title = title;

        button.style.position = 'fixed';
        button.style.right = '20px';
        button.style.zIndex = '2147483647';

        button.style.padding = '8px 12px';
        button.style.border = '1px solid #aaa';
        button.style.borderRadius = '5px';

        button.style.background = '#fff';
        button.style.cursor = 'pointer';

        button.addEventListener('click', handler);

        return button;
    }

    function addButtons() {
        if (document.querySelector('#dmed-rx-save-btn')) {
            return;
        }

        const saveBtn = createButton(
            '💾 Запомнить рецепт',
            'Запомнить рецепт из текущей записи',
            rememberPreviousRecipe
        );

        saveBtn.id = 'dmed-rx-save-btn';
        saveBtn.style.bottom = '105px';

        const applyBtn = createButton(
            '↻ Предыдущий рецепт',
            'Заполнить новый рецепт данными предыдущего',
            applyPreviousRecipe
        );

        applyBtn.id = 'dmed-rx-apply-btn';
        applyBtn.style.bottom = '65px';

        document.body.appendChild(saveBtn);
        document.body.appendChild(applyBtn);
    }

    // ------------------------------------------------------------
    // Старт
    // ------------------------------------------------------------

    function init() {
        addButtons();

        log(
            'Готов.',
            'Сохранённые рецепты:',
            getStoredRecipes()
        );

        // DMED постоянно перерисовывает интерфейс.
        // Если наши кнопки исчезли — возвращаем их.
        setInterval(() => {
            addButtons();
        }, 1000);
    }

    if (
        document.readyState === 'loading'
    ) {
        document.addEventListener(
            'DOMContentLoaded',
            init
        );
    } else {
        init();
    }

})();
