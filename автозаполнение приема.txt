(function () {
'use strict';

const C={
 schedule:'787583',
 service:'29791981',
 treatment1:'5',
 treatment1Name:'Динамическое наблюдение',
 treatment2:'47',
 treatment2Name:'Динамическое наблюдение с социально-значимыми заболеваниями',
 finance:'67',
 diagnosisCode:'F20.0',
 diagnosisName:'Параноидная шизофрения',
 sickId:'2526',
 diagnosisType:'5',
 diagType:'2',
 charSick:'4',
 narcService:'29791982',
 narcDiagnosisCode:'F10.2',
 narcDiagnosisName:'Психические и поведенческие расстройства, вызванные употреблением алкоголя, синдром зависимости',
 narcSickId:'2418'
};

let MODE='psy';

function cfg(){
 return MODE==='narc'
  ? {
     ...C,
     service:C.narcService,
     diagnosisCode:C.narcDiagnosisCode,
     diagnosisName:C.narcDiagnosisName,
     sickId:C.narcSickId
    }
  : C;
}

const sleep=ms=>new Promise(r=>setTimeout(r,ms));
const log=(...x)=>console.log('[Damumed Автозаполнение]',...x);

async function waitUntil(test,timeout=4000,interval=50){
 const end=Date.now()+timeout;

 while(Date.now()<end){
  try{
   const value=test();
   if(value)return value;
  }catch(e){}

  await sleep(interval);
 }

 return null;
}

function wins(){
 return typeof Ext!=='undefined'&&Ext.ComponentQuery
  ? Ext.ComponentQuery.query('window'):[];
}

function field(w,n){
 return w?.query('field').find(f=>f.name===n)||null;
}

function byLabel(w,s){
 s=String(s).toLowerCase();
 return w?.query('field').find(
  f=>String(f.fieldLabel||'').toLowerCase().includes(s)
 )||null;
}

function booking(){
 return wins().find(
  w=>String(w.title||'').trim()==='Запись на прием'
 )||null;
}

function patient(){
 return wins().find(
  w=>w.xtype==='window'&&!!field(w,'diagName')
 )||null;
}

async function waitStore(c,n=40){
 const st=c?.getStore?.();
 if(!st)return null;

 for(let i=0;i<n;i++){
  if(st.getCount()>0)return st;
  await sleep(250);
 }

 return st;
}

function rec(c,id){
 const st=c?.getStore?.();
 if(!st)return null;

 return st.getRange().find(
  r=>String(r.get('id')??r.get('ID'))===String(id)
 )||null;
}

function select(c,id){
 if(!c)return false;

 const r=rec(c,id);
 if(!r)return false;

 try{
  c.select(r);
 }catch(e){
  c.setValue(id);
 }

 try{
  c.fireEvent('select',c,[r]);
 }catch(e){}

 try{
  c.fireEvent('change',c,id,null);
 }catch(e){}

 return String(c.getValue())===String(id);
}

async function fillSchedule(w){
 const f=field(w,'GridScheduleID');
 if(!f)return;

 await waitStore(f);

 log(
  select(f,C.schedule)
   ? '✓ График'
   : '⚠ График не найден'
 );

 await sleep(300);
}

 log('✓ Повторный');
}

async function fillTreatment(w){
 const n1=await waitUntil(()=>{
  const value=field(w,'node1');
  return value?.objMask?value:null;
 },3000);

 if(!n1){
  log('⚠ Повод обращения: node1/objMask не найден');
  return false;
 }

 const m=n1.objMask;

 const r1={
  id:C.treatment1,
  nameRU:C.treatment1Name
 };

 const r2={
  id:C.treatment2,
  nameRU:C.treatment2Name
 };

 log('→ Заполняем повод обращения');

 try{
  if(typeof m.fillingLevelStore==='function'){
   m.fillingLevelStore(true,[r1,r2],2);
  }else{
   throw new Error('fillingLevelStore отсутствует');
  }
 }catch(e){
  try{
   m.clearCombo(-1);
   m.rec=[];
   m.fillingLevel=true;
   m.isFilling=true;

   const c1=m.addCombo(0,0);

   if(c1){
    c1.getStore().loadData([r1]);
    c1.noRefresh=true;
    c1.setValue(C.treatment1);
    c1.noRefresh=false;
   }

   const c2=m.addCombo(C.treatment1,1);

   if(c2){
    c2.getStore().loadData([r2]);
    c2.noRefresh=true;
    c2.setValue(C.treatment2);
    c2.noRefresh=false;
   }

   m.fillingLevel=false;
   m.isFilling=false;
  }catch(x){
   log('❌ Ошибка повода обращения:',x);
   return false;
  }
 }

 // fillingLevelStore создаёт node2 асинхронно. Каждый раз читаем
 // компоненты заново, а не используем устаревший null.
 const ready=await waitUntil(()=>{
  const first=field(w,'node1');
  const second=field(w,'node2');
  return first&&second?{first,second}:null;
 },3500);

 if(!ready){
  log('⚠ Повод обращения: второй список не успел создаться');
  return false;
 }

 const hidden=field(w,'TreatmentReasonID');

 // Подстраховываем оба видимых combo и скрытое итоговое поле.
 try{
  if(String(ready.first.getValue?.())!==C.treatment1){
   ready.first.setValue(C.treatment1);
  }

  if(String(ready.second.getValue?.())!==C.treatment2){
   const store=ready.second.getStore?.();
   if(store&&!rec(ready.second,C.treatment2))store.loadData([r2]);
   ready.second.setValue(C.treatment2);
  }

  if(hidden)hidden.setValue(C.treatment2);
  if(m.objValue)m.objValue.setValue(C.treatment2);
  m.rawData=C.treatment2;

  ready.second.fireEvent?.('select',ready.second,[rec(ready.second,C.treatment2)||r2]);
  ready.second.fireEvent?.('change',ready.second,C.treatment2,null);
  hidden?.fireEvent?.('change',hidden,C.treatment2,null);
  m.objValue?.fireEvent?.('change',m.objValue,C.treatment2,null);
  m.validationCheck?.();
 }catch(e){
  log('❌ Ошибка фиксации повода обращения:',e);
  return false;
 }

 const ok=await waitUntil(()=>{
  const first=field(w,'node1');
  const second=field(w,'node2');
  const result=field(w,'TreatmentReasonID');

  return String(first?.getValue?.())===C.treatment1&&
   String(second?.getValue?.())===C.treatment2&&
   String(result?.getValue?.())===C.treatment2;
 },1500);

 log(
  ok?'✓ Повод обращения заполнен':'⚠ Повод обращения не прошёл проверку',
  'node1=',field(w,'node1')?.getValue?.(),
  'node2=',field(w,'node2')?.getValue?.(),
  'TreatmentReasonID=',field(w,'TreatmentReasonID')?.getValue?.()
 );

 return !!ok;
}

async function fillFinance(w){
 const f=field(w,'FinanceSourceID');

 if(!f){
  log('⚠ FinanceSourceID не найден');
  return false;
 }

 try{
  if(typeof w.financeSourceInit==='function'){
   w.financeSourceInit();
  }
 }catch(e){}

 await sleep(200);
 await waitStore(f);

 if(!rec(f,C.finance)){
  log('⚠ Источник финансирования 67 не найден');
  return false;
 }

 let ok=select(f,C.finance);

 if(!ok){
  f.setValue(C.finance);
 }

 try{
  f.fireEvent('change',f,C.finance,null);
 }catch(e){}

 await sleep(300);

 log(
  '✓ Источник финансирования:',
  f.getValue(),
  f.getRawValue()
 );

 return String(f.getValue())===C.finance;
}

async function fillBooking(w){
 await fillSchedule(w);
 await fillService(w);
 fillReadmission(w);
 await fillFinance(w);

 // Источник финансирования перестраивает зависимые поля ExtJS,
 // поэтому повод обращения обязательно заполняем последним.
 await fillTreatment(w);

 // Даём последним AJAX/listener-обработчикам Damumed завершиться.
 // Если компонент был пересоздан и значение исчезло — восстанавливаем.
 await sleep(350);

 const treatmentOk=
  String(field(w,'node1')?.getValue?.())===C.treatment1&&
  String(field(w,'node2')?.getValue?.())===C.treatment2&&
  String(field(w,'TreatmentReasonID')?.getValue?.())===C.treatment2;

 if(!treatmentOk){
  log('→ Damumed перерисовал повод обращения, заполняем повторно');
  await fillTreatment(w);
 }

 log('✓ Первое окно заполнено');
}

   field(w,'DiagnosisTypeID')||

  log('⚠ Тип диагноза не найден');
  return false;
 }

 await waitStore(f);

 const store=f.getStore?.();

 if(!store){
  log('⚠ Store диагноза не найден');
  return false;
 }

 const records=store.getRange();

 const target=records.find(r=>
  String(r.get('ID'))==='5'

  return false;
 }

 f.setValue(target.get('ID'));

  f.fireEvent('change',f,target.get('ID'),null);
 }catch(e){}

 await sleep(300);

 log(
  '✓ Диагноз:',
  target.get('DisplayName'),
  'ID='+target.get('ID')
 );

   byLabel(w,'Источник фин.');

  log('⚠ Второе окно: источник финансирования не найден');
  return false;

  const refresh=

    [...root.querySelectorAll('a')].find(a=>

        c=>c.includes('image-button-refresh')

   .replace(/\s+/g,' ')
   .trim()
   .includes(targetText)

  log('⚠ Не найден вариант: '+targetText);

  '✓ Источник финансирования: '+

function checkSave(w){
 const p=w.down('#frm');
 const b=w.down('#btnSave');

 if(!p||!b)return;

 const form=p.getForm();
 form.isValid();

 if(!b.disabled){
  log('✓ Сохранить активна');
  return;
 }

 const bad=[];

 form.getFields().each(f=>{
  if(f.disabled||f.hidden)return;

  let ok=true;

  try{
   ok=f.isValid();
  }catch(e){}

  if(!ok){
   bad.push(
    f.fieldLabel||
    f.name||
    f.itemId||
    'поле'
   );
  }
 });

 log(
  bad.length
   ? '⚠ Форма невалидна: '+bad.join(', ')
   : '⚠ Сохранить пока не активна'
 );
}

function hideRecommendationPopup(){
 const elements=document.querySelectorAll('[role="dialog"]');

 elements.forEach(el=>{
  const text=(el.textContent||'')
   .replace(/\s+/g,' ')
   .trim();

  if(
   text.includes('Вы можете выделить текст рекомендации')&&
   text.includes('Добавить как рекомендацию по лечению')
  ){
   el.style.setProperty('display','none','important');
  }
 });
}

async function autoFill(mode='psy'){
 MODE=mode;

 if(window.__DAMUMED_RUNNING)return;

 window.__DAMUMED_RUNNING=true;

 try{
  const b=booking();

  if(b){
   await fillBooking(b);
  }else{
   log('⚠ Окно «Запись на прием» не найдено');
  }

  let p=null;

  for(let i=0;i<60;i++){
   p=patient();
   if(p)break;
   await sleep(300);
  }

  if(p){
   await fillPatient(p);
  }else{
   log('⚠ Второе окно приёма не найдено');
  }
 }catch(e){
  console.error('[Damumed Автозаполнение] ❌',e);
 }finally{
  window.__DAMUMED_RUNNING=false;
 }
}

function positionButton(){
 const b=document.getElementById('damumed-auto-fill-btn');
 if(!b)return;

 const w=booking();

 if(!w){
  b.style.display='none';
  return;
 }

 let save=w.down('#btnSave');

 if(!save){
  save=w.query('button').find(
   x=>String(x.text||'')
    .toLowerCase()
    .includes('сохран')
  );
 }

 if(!save?.getEl?.()?.dom){
  b.style.display='none';
  return;
 }

 const r=save.getEl().dom.getBoundingClientRect();

 b.style.display='block';
 b.style.left=Math.max(5,r.left-180)+'px';
 b.style.top=Math.max(5,r.top)+'px';
}

 createButton();
 createNarcButton();

 setInterval(()=>{
  positionButton();
  positionNarcButton();
 },500);

 new MutationObserver(()=>{
  positionButton();
  positionNarcButton();
 }).observe(document.body,{
  childList:true,
  subtree:true
 });

 positionButton();
 positionNarcButton();

 log('✓ CLEAN FIX 8.1 запущен');
}

const timer=setInterval(()=>{
 if(typeof Ext!=='undefined'&&Ext.ComponentQuery){
  clearInterval(timer);
  start();
 }
},500);

new MutationObserver(mutations=>{
 mutations.forEach(mutation=>{
  mutation.addedNodes.forEach(node=>{
   if(node.nodeType!==1)return;

   const elements=[];

   if(
    node.matches?.(
     '[role="dialog"], .modal, [class*="modal"], [class*="popup"], [class*="dialog"]'
    )
   ){
    elements.push(node);
   }

   if(node.querySelectorAll){
    elements.push(
     ...node.querySelectorAll(
      '[role="dialog"], .modal, [class*="modal"], [class*="popup"], [class*="dialog"]'
     )
    );
   }

   elements.forEach(el=>{
    const text=(el.textContent||'')
     .replace(/\s+/g,' ')
     .trim();

    if(
     text.includes('Вы можете выделить текст рекомендации')&&
     text.includes('Добавить как рекомендацию по лечению')
    ){
     el.style.setProperty('display','none','important');
    }
   });
  });
 });
}).observe(document.body,{
 childList:true,
 subtree:true
});

})();
